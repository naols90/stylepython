ስላይድ 6 🌈 Colorama – በጣም ቀላል ክሮስ-ፕላትፎርም ዘዴ

from colorama import Fore, Back, Style, init
init(autoreset=True) # ከእያንዳንዱ መስመር በኋላ ነባሪ ያደርጋል

print(Fore.RED + "ይህ ቀይ ጽሑፍ ነው")
print(Fore.GREEN + "አረንጓዴ ስኬት!" + Style.RESET_ALL)
print(Back.YELLOW + Fore.BLACK + "ጥንቃቄ: ቢጫ ዳራ ጥቁር ጽሑፍ")

from termcolor import colored

text = colored("ሰላም ፓይተን!", color="cyan", on_color="on_magenta", attrs=["bold"])
print(text)
print(colored("ስኬት", "green", attrs=["reverse"]))
print(colored("ስህተት", "red", "on_white"))
  • color="red" ወይም "blue", "yellow"
  • on_color="on_red" ለዳራ ቀለም
  • attrs=["bold", "underline"] ውፍረት ወይም መስመር ስር
ስላይድ 8 📝 ተግባራዊ ምሳሌ – ማስጀመሪያ ስክሪፕት በቀለም
def print_status(msg, status="info"):
    if status == "success":
        print(f"\033[92m✅ {msg}\033[0m")
    elif status == "error":
        print(f"\033[91m❌ {msg}\033[0m")
    elif status == "warning":
        print(f"\033[93m⚠️ {msg}\033[0m")
    else:
        print(f"\033[94mℹ️ {msg}\033[0m")

print_status("ፋይል በተሳካ ሁኔታ ተቀመጠ", "success")
print_status("የአውታረ መረብ ግንኙነት አልተገኘም", "error")
print_status("ማህደረ ትውስታ እየቀነሰ ነው", "warning")
